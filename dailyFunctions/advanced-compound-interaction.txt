    public static float analyzeCompoundInteraction(int compoundStructure, float[] concentrationLevels, int experimentDuration) {
        final double E = Math.E;
        final double PI = Math.PI;

        // Decode compound structure
        int molecularWeight = compoundStructure & 0xFF;
        int polarity = (compoundStructure >> 8) & 0xF;
        int reactivity = (compoundStructure >> 12) & 0xF;

        // Model interaction over time
        double timeEffect = Math.sin(experimentDuration / 365.0 * 2 * PI) + Math.cos(experimentDuration / 365.0 * PI);

        // Simulate chemical reactions
        int reactionProduct = ((compoundStructure << 2) ^ (compoundStructure >> 2)) & 0xFFFF;

        // Model growth/decay
        double growthFactor = Math.log(1 + experimentDuration) / Math.log(366);

        // Custom algorithm using XOR
        int mixedProperties = molecularWeight ^ polarity ^ reactivity;

        // Calculate stability using GCD
        int stability = gcd(molecularWeight, mixedProperties);

        // Calculate interaction score
        double rawScore = 0;
        for (float concentration : concentrationLevels) {
            rawScore += concentration * reactionProduct * timeEffect * growthFactor * stability;
        }

        // Normalize score to 0-100 range
        double normalizedScore = rawScore / (concentrationLevels.length * 65535 * 2 * Math.log(366) * 255) * 100;

        // Ensure score is between 0 and 100
        return (float) Math.max(0, Math.min(100, normalizedScore));
    }

    private static int gcd(int a, int b) {
        if (b == 0) return a;
        return gcd(b, a % b);
    }
